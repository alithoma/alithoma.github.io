2 Salin Kode Google Apps Script Ini

Buka Extensions > Apps Script di Google Sheet Anda, hapus semua kode lama, lalu tempel kode berikut:

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Data_Siswa");
  var data = sheet.getDataRange().getValues();
  var students = [];
  
  // Mulai dari baris ke-2 (index 1) untuk melewati header
  for (var i = 1; i < data.length; i++) {
    var status = data[i][3] ? data[i][3].toString().trim().toLowerCase() : "aktif";
    if (data[i][1] && status === "aktif") {
      students.push({
        id: data[i][0].toString(),
        name: data[i][1].toString(),
        program: data[i][2].toString()
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    students: students
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Rekap_Presensi");
    var payload = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    
    payload.records.forEach(function(item) {
      sheet.appendRow([
        timestamp,
        payload.date,
        payload.program,
        payload.mentor,
        payload.topic,
        item.id,
        item.name,
        item.status,
        item.note
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Presensi berhasil disimpan!"
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
