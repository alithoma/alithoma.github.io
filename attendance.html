Integrasi Google Sheets

Hubungkan aplikasi ini langsung dengan spreadsheet Xpert Academy Anda secara gratis tanpa pihak ketiga.

URL Web App Google Apps Script

Tempelkan URL deployment dari Google Apps Script Anda di bawah ini agar data siswa otomatis terbaca dan presensi langsung masuk ke Google Sheet.

Struktur Tab di Google Sheet Anda:

  • Tab 1 bernama Data_Siswa (Diisi Admin): Kolom A = ID Siswa, Kolom B = Nama Siswa.
  • Tab 2 bernama Log_Presensi (Terisi Otomatis oleh Web App ini).

Panduan & Kode Google Apps Script

Salin kode ini ke menu Extensions > Apps Script di Google Sheet Anda.

// KODE GOOGLE APPS SCRIPT UNTUK XPERT ACADEMY
// 1. Buat Google Sheet dengan 2 tab: "Data_Siswa" dan "Log_Presensi"
// 2. Di tab "Data_Siswa", buat header baris 1: ID_Siswa | Nama_Siswa
// 3. Klik Extensions > Apps Script, tempel kode ini, lalu klik Deploy > New Deployment > Web App
// 4. Set "Execute as: Me" dan "Who has access: Anyone", lalu salin URL Web App-nya.

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const action = e.parameter.action || 'getStudents';
  
  if (action === 'getStudents') {
    const sheet = ss.getSheetByName('Data_Siswa');
    if (!sheet) return responseJSON({ status: 'error', message: 'Tab Data_Siswa tidak ditemukan' });
    
    const data = sheet.getDataRange().getValues();
    const students = [];
    for (let i = 1; i < data.length; i++) {
      if (data[i][1]) { // Pastikan nama tidak kosong
        students.push({
          id: data[i][0] || 'XA-' + i,
          name: data[i][1]
        });
      }
    }
    return responseJSON({ status: 'success', data: students });
  }
  
  if (action === 'getHistory') {
    const sheet = ss.getSheetByName('Log_Presensi');
    if (!sheet) return responseJSON({ status: 'success', data: [] });
    const data = sheet.getDataRange().getValues();
    const history = [];
    for (let i = data.length - 1; i >= 1 && history.length < 30; i--) {
      history.push({
        timestamp: data[i][0],
        date: data[i][1],
        mentor: data[i][2],
        students: data[i][3],
        materialTitle: data[i][5],
        materialDesc: data[i][6],
        notes: data[i][7]
      });
    }
    return responseJSON({ status: 'success', data: history });
  }
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    let sheet = ss.getSheetByName('Log_Presensi');
    
    // Buat tab Log_Presensi otomatis jika belum ada
    if (!sheet) {
      sheet = ss.insertSheet('Log_Presensi');
      sheet.appendRow([
        'Waktu Input', 'Tanggal Kelas', 'Mentor', 
        'Siswa Hadir', 'Jumlah Hadir', 'Judul Materi', 'Detail Materi', 'Catatan'
      ]);
      sheet.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#e2e8f0');
    }
    
    const payload = JSON.parse(e.postData.contents);
    const studentNames = payload.students.map(s => s.name + ' (' + s.id + ')').join(', ');
    
    sheet.appendRow([
      new Date(),
      payload.date,
      payload.mentor,
      studentNames,
      payload.students.length,
      payload.materialTitle,
      payload.materialDesc,
      payload.notes || '-'
    ]);
    
    return responseJSON({ status: 'success', message: 'Presensi berhasil disimpan ke Google Sheet!' });
  } catch (err) {
    return responseJSON({ status: 'error', message: err.toString() });
  }
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
