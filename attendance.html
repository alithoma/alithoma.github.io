Integrasi Google Sheets

Salin kode Apps Script berikut ke Google Sheet Anda

Langkah Setup Google Sheet:

  1. Buat Google Sheet baru dengan 2 tab (sheet):
    • Tab 1 bernama DataSiswa (Kolom A: ID Siswa, Kolom B: Nama Siswa, Kolom C: Program). Isi daftar nama siswa di sini.
    • Tab 2 bernama Presensi (Header otomatis dibuat oleh script).
  2. Klik menu Ekstensi (Extensions) > Apps Script.
  3. Hapus semua kode yang ada, lalu tempel kode di bawah ini dan klik Simpan (Save).
  4. Klik Terapkan (Deploy) > Deployment baru (New deployment) > Pilih jenis Aplikasi Web (Web app).
  5. Set Yang memiliki akses (Who has access) menjadi Siapa saja (Anyone), lalu klik Terapkan.
  6. Salin URL Aplikasi Web yang muncul, lalu tempel ke variabel SCRIPT_URL di baris paling atas bagian <script> file HTML ini.
Kode Google Apps Script (Code.gs) 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSiswa = ss.getSheetByName("DataSiswa");
  
  if (!sheetSiswa) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error", 
      message: "Tab 'DataSiswa' tidak ditemukan!"
    })).setMimeType(ContentService.MimeType.JSON);
  }
  
  var data = sheetSiswa.getDataRange().getValues();
  var students = [];
  
  // Mulai dari baris ke-2 (index 1) melewati header
  for (var i = 1; i < data.length; i++) {
    if (data[i][1] !== "") { // Pastikan nama siswa tidak kosong
      students.push({
        id: data[i][0] ? data[i][0].toString() : "XA-" + i,
        nama: data[i][1].toString(),
        program: data[i][2] ? data[i][2].toString() : "Coding & Robotics"
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    students: students
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetPresensi = ss.getSheetByName("Presensi");
    
    // Buat tab Presensi otomatis jika belum ada
    if (!sheetPresensi) {
      sheetPresensi = ss.insertSheet("Presensi");
      sheetPresensi.appendRow([
        "Timestamp", "Tanggal Pertemuan", "ID Siswa", 
        "Nama Siswa", "Status Kehadiran", 
        "Materi Diajarkan", "Nama Mentor"
      ]);
      sheetPresensi.getRange("A1:G1").setFontWeight("bold").setBackground("#e0e7ff");
    }
    
    var payload = JSON.parse(e.postData.contents);
    
    sheetPresensi.appendRow([
      new Date(),
      payload.tanggal,
      payload.studentId,
      payload.studentName,
      payload.status,
      payload.materi,
      payload.mentor
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      message: "Data presensi berhasil disimpan!"
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({
      status: "error",
      message: error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
