Panduan Integrasi Google Sheets & GitHub Pages

Ikuti 3 langkah mudah untuk menghubungkan database siswa dan rekap presensi

1 Siapkan Struktur Google Sheets

Buat file Google Spreadsheet baru, lalu buat 2 Tab Sheet di bagian bawah dengan nama persis seperti berikut:

Tab 1: DataSiswa

Isi header baris ke-1 (Kolom A sampai D):

ID_Siswa | Nama_Siswa | Program | Kelas

Tab 2: RekapPresensi

Header otomatis dibuat oleh skrip, atau isi baris ke-1:

Timestamp | Tanggal | Materi_Ajar | Mentor | Program | ID_Siswa | Nama_Siswa | Status | Catatan

2 Salin Kode Google Apps Script

Di Google Sheets, klik menu Ekstensi (Extensions) → Apps Script. Hapus kode lama, tempel kode di bawah ini, lalu klik Deploy → New deployment → Web app (Set Execute as: Me dan Who has access: Anyone). 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("DataSiswa");
  if (!sheet) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sheet DataSiswa tidak ditemukan" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var data = sheet.getDataRange().getValues();
  var students = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][1]) {
      students.push({
        id: String(data[i][0] || "XP-" + i),
        name: String(data[i][1]),
        program: String(data[i][2] || ""),
        level: String(data[i][3] || "Reguler")
      });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", students: students }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("RekapPresensi");
    if (!sheet) {
      sheet = ss.insertSheet("RekapPresensi");
      sheet.appendRow(["Timestamp", "Tanggal", "Materi_Ajar", "Mentor", "Program", "ID_Siswa", "Nama_Siswa", "Status", "Catatan"]);
    }
    var payload = JSON.parse(e.postData.contents);
    var timestamp = new Date();
    var records = payload.records || [];
    
    records.forEach(function(item) {
      sheet.appendRow([
        timestamp,
        payload.date,
        payload.materiAjar,
        payload.mentor,
        payload.program,
        item.id,
        item.name,
        item.status,
        item.note || "-"
      ]);
    });
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: records.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

3 Tempel URL ke Kode & Upload ke GitHub Pages

  1. Salin Web App URL (berakhiran /exec) dari hasil Deploy Apps Script.
  2. Buka file index.html ini, temukan konstanta GOOGLE_SHEET_API_URL di bagian paling atas tag <script>, lalu ganti teks di dalamnya dengan URL Anda.
  3. Upload index.html ke repository GitHub Anda, lalu aktifkan Settings → Pages → Deploy from branch (main).
