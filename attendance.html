Panduan Integrasi Google Sheets

1. Siapkan 2 Tab (Sheet) di Google Sheets Anda:

  • Tab 1 beri nama Siswa → Kolom A: ID, Kolom B: Nama Siswa, Kolom C: Program (Coding/Robotics/STEAM), Kolom D: Level/Kelas. (Diisi oleh Admin)
  • Tab 2 beri nama Presensi → Kolom A s/d H: Timestamp, Tanggal, ID Siswa, Nama Siswa, Program, Status, Materi, Catatan, Mentor.

2. Salin Kode Google Apps Script ini:

Buka Google Sheets → Ekstensi → Apps Script → Tempel kode di bawah → Klik Terapkan (Deploy) → Deployment Baru → Jenis: Aplikasi Web → Akses: Siapa saja (Anyone).

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSiswa = ss.getSheetByName("Siswa");
  
  if (!sheetSiswa) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sheet 'Siswa' tidak ditemukan" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var data = sheetSiswa.getDataRange().getValues();
  var students = [];
  
  // Mulai dari baris ke-2 (index 1) untuk melewati header
  for (var i = 1; i < data.length; i++) {
    if (data[i][1]) { // Pastikan nama siswa tidak kosong
      students.push({
        id: data[i][0] || "STD-" + i,
        nama: data[i][1],
        program: data[i][2] || "Coding",
        level: data[i][3] || "Basic"
      });
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: students }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetPresensi = ss.getSheetByName("Presensi");
    
    // Otomatis buat sheet Presensi jika belum ada
    if (!sheetPresensi) {
      sheetPresensi = ss.insertSheet("Presensi");
      sheetPresensi.appendRow(["Timestamp", "Tanggal", "ID Siswa", "Nama Siswa", "Program", "Status", "Materi", "Catatan", "Mentor"]);
    }

    var params = JSON.parse(e.postData.contents);
    
    sheetPresensi.appendRow([
      new Date(),
      params.tanggal,
      params.idSiswa,
      params.namaSiswa,
      params.program,
      params.status,
      params.materi,
      params.catatan,
      params.mentor
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Data tersimpan" }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
